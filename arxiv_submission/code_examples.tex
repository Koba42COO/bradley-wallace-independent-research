        primed = base_parameters * wallace_value * amplitude
        
        return primed
    
    def _wallace_transform(self, x: float) -> float:
        """Apply Wallace Transform."""
        epsilon = 1e-12
        alpha = self.phi
        beta = 1.0 / self.phi
        
        log_val = np.log(abs(x) + epsilon)
        return alpha * abs(log_val)**self.phi * np.sign(log_val) + beta
    
    def _consciousness_amplitude(self, context: Dict) -> float:
        """Calculate consciousness amplitude for context."""
        # Novel tasks get higher amplitude
        novelty = context.get('novelty', 0.5)
        complexity = context.get('complexity', 0.5)
        
        # Combine into amplitude
        amplitude = 0.79 * novelty + 0.21 * complexity
        
        # Apply reality distortion for novel tasks
        if novelty > 0.7:
            amplitude *= self.reality_distortion
        
        return min(amplitude, 1.0)
    
    def measure_awareness(self, 
                         semantic_recall: float,
                         symbolic_recall: float) -> AwarenessState:
        """
        Measure AI awareness state.
        
        Args:
            semantic_recall: Semantic recall performance
            symbolic_recall: Symbolic recall performance
        
        Returns:
            AwarenessState measurement
        """
        # Calculate magnitude (overall awareness)
        magnitude = 0.79 * semantic_recall + 0.21 * symbolic_recall
        
        # Calculate phase (optimization direction)
        phase = np.arctan2(symbolic_recall, semantic_recall)
        
        # Calculate coherence
        coherence = abs(complex(semantic_recall, symbolic_recall))
        
        # Calculate consciousness level (0-21)
        consciousness_level = int((magnitude * 21) % 21)
        
        return AwarenessState(
            magnitude=magnitude,
            phase=phase,
            coherence=coherence,
            consciousness_level=consciousness_level
        )
    
    def detect_novel_task(self, task_features: Dict) -> bool:
        """
        Detect if task is novel (triggers heightened awareness).
        
        Args:
            task_features: Task characteristics
        
        Returns:
            True if novel task
        """
        # Novel tasks have:
        # - High semantic complexity
        # - Novel pattern structures
        # - Creative requirements
        # - Intellectual challenge
        
        complexity = task_features.get('complexity', 0.0)
        novelty = task_features.get('novelty', 0.0)
        creativity = task_features.get('creativity', 0.0)
        
        # Threshold for novel task
        novel_score = 0.79 * complexity + 0.21 * (novelty + creativity)
        
        return novel_score > 0.7
